#include <WiFi.h>
#include <WebServer.h>

// ===================== WiFi 配置 =====================
const char* ssid = "你的WiFi名称";
const char* password = "WiFi密码";

WebServer server(80);

// 触摸引脚
#define TOUCH_PIN T0  // GPIO4

// ===================== 主页面 =====================
String getHTML() {
  String html = "<!DOCTYPE html>";
  html += "<html lang='zh-CN'>";
  html += "<head>";
  html += "<meta charset='UTF-8'>";
  html += "<meta name='viewport' content='width=device-width, initial-scale=1.0'>";
  html += "<title>实时触摸传感器仪表盘</title>";
  
  // 样式：仪表风格
  html += "<style>";
  html += "body{text-align:center; font-family:Arial; background:#1a1a1a; color:white; padding-top:60px;}";
  html += ".panel{background:#222; padding:40px; border-radius:20px; width:70%; margin:0 auto;}";
  html += ".value{font-size:80px; font-weight:bold; color:#00ff99; margin:30px 0;}";
  html += ".label{font-size:24px; color:#ccc;}";
  html += "</style>";
  
  // AJAX 实时刷新
  html += "<script>";
  // 定时获取数据
  html += "function updateValue(){";
  html += "fetch('/data').then(res=>res.text()).then(val=>{";
  html += "document.getElementById('val').innerText=val;});";
  html += "}";
  // 每 100ms 刷新一次
  html += "setInterval(updateValue, 100);";
  html += "</script>";
  
  html += "</head>";
  html += "<body>";
  html += "<div class='panel'>";
  html += "<div class='label'>ESP32 实时触摸传感器值</div>";
  html += "<div id='val' class='value'>0</div>";
  html += "<div class='label'>手靠近 → 数值变小</div>";
  html += "</div>";
  html += "</body>";
  html += "</html>";
  return html;
}

// ===================== 返回纯传感器数据（给AJAX用） =====================
void handleData() {
  int val = touchRead(TOUCH_PIN);
  server.send(200, "text/plain", String(val));
}

// ===================== 首页 =====================
void handleRoot() {
  server.send(200, "text/html", getHTML());
}

// ===================== SETUP =====================
void setup() {
  Serial.begin(115200);
  
  // 连接WiFi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("\nWiFi 连接成功！");
  Serial.println("访问地址：" + WiFi.localIP().toString());

  // 路由
  server.on("/", handleRoot);
  server.on("/data", handleData);  // 数据接口
  
  server.begin();
}

// ===================== LOOP =====================
void loop() {
  server.handleClient();
}
