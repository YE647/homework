// ===================== 引脚定义 =====================
#define LED_A_PIN  2    // 第一个LED引脚 (板载LED/外接LED)
#define LED_B_PIN  4   // 第二个LED引脚 (必须是PWM引脚)

// ===================== 呼吸灯参数 =====================
int brightnessA = 0;    // LED A 亮度
int brightnessB = 255;  // LED B 初始反向亮度
int fadeStep = 3;       // 渐变速度（数值越大越快）

void setup() {
  // 设置两个LED为输出模式
  pinMode(LED_A_PIN, OUTPUT);
  pinMode(LED_B_PIN, OUTPUT);

  // 初始点亮状态（反向）
  analogWrite(LED_A_PIN, brightnessA);
  analogWrite(LED_B_PIN, brightnessB);
}

void loop() {
  // ===================== 核心：双灯反向渐变 =====================
  // LED A 递增变亮
  brightnessA += fadeStep;
  // LED B 递减变暗 (完美反向)
  brightnessB -= fadeStep;

  // 边界判断：到达极值后反转方向
  if (brightnessA >= 255 || brightnessA <= 0) {
    fadeStep = -fadeStep; // 渐变方向反转
  }

  // ===================== PWM输出亮度 =====================
  analogWrite(LED_A_PIN, brightnessA);
  analogWrite(LED_B_PIN, brightnessB);

  // 小延时保证平滑过渡
  delay(15);
}
